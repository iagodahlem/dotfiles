# Debian family packages (Debian, Raspberry Pi OS, Ubuntu), installed in one call by install_apt.
# scripts/install-apt-repos.sh adds the third-party sources first (docker, tailscale, eza, docker-ctop).
# A package with no candidate on the running release is skipped with a warning.
# Not listed here: mise (mise.run installer) and atuin and procs (installed through mise),
# all three handled by scripts/install-mise.sh, and powerlevel10k, which Debian and Ubuntu do not package (scripts/install-shell.sh clones it).
# Trailing comments are stripped by read_list_items in scripts/install-packages.sh.

# --- base ---
ca-certificates
curl
locales
sudo
tzdata

# --- CLI core ---
bat             # cat with syntax highlighting, installs as batcat
eza             # ls replacement, from the eza repository
fd-find         # find replacement, installs as fdfind
fzf             # fuzzy finder
jq
less
mosh            # roaming SSH
neovim
ripgrep         # grep replacement
tmux

# --- git and shell ---
gh              # GitHub CLI
git
git-delta       # diff pager
zsh
zsh-autosuggestions      # fish-like inline suggestions
zsh-syntax-highlighting  # command syntax coloring

# --- system tools ---
bind9-dnsutils  # dig and nslookup (dnsutils is a virtual name for it on Debian 13)
btop            # system monitor
docker-ctop     # container metrics, from the Azlux repository
duf             # disk usage overview
fastfetch       # system info, Debian 13 and newer
ncdu            # disk usage browser
speedtest-cli   # internet speed test
xclip           # X11 clipboard tool (tmux copy-pipe)

# --- containers and network ---
containerd.io   # from the Docker repository, as are the docker packages below
docker-buildx-plugin
docker-ce
docker-ce-cli
docker-compose-plugin
tailscale       # mesh VPN, from the Tailscale repository
